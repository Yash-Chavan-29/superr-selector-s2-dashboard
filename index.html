Core Mathematical Architecture

I_bat = (Runs_bat - xRuns_bat) - V_w * (1_out - xWkt_bat)
I_bowl = (xRuns_bowl - Runs_bowl) + V_w * (1_wkt - xWkt_bowl)
HP_EB = 100 * Sum(w_i * I_i) / (Sum(w_i) + 120)
Resil = HP_EB - max(0, Norm_EB) [Ceiling-Safe Delta]
PRR = 50 + 15 * (0.60*z_HP + 0.20*z_Resil + 0.20*z_Cons)
